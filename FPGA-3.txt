//Initialize
module wiseboi(m,a,b);
input [3:0]a,b;
output [7:0]m ;
wire [3:0]p0, p1, p2, p3;
wire s5, s4,s3,s2,s1,s0;
wire c5, c4,c3,c2,c1,c0;
wire k1,k2,l1,l2;
wire n4,n3,n2,n1,n0;
//Generate Partial Products
assign p0 = a & {4{b[0]}};
assign p1 = a & {4{b[1]}};
assign p2 = a & {4{b[2]}};
assign p3 = a & {4{b[3]}};
halfadd h1(k1,l1,p2[1],p3[0]);
halfadd h2(k2,l2,p2[2],p3[1]);
halfadd h3(s0,c0,p0[1],p1[0]);
fulladd f1(s1,c1,p0[2],p1[1],p2[0]);
fulladd f2(s2,c2,p0[3],p1[2],k1);
fulladd f3(s3,c3,p1[3],k2,l1);
fulladd f4(s4,c4,p2[3],p3[2],l2);
halfadd h4(m[2],n0,s1,c0);
fulladd f5(m[3],n1,s2,c1,n0);
fulladd f6(m[4],n2,s3,c2,n1);
fulladd f7(m[5],n3,s4,c3,n2);
fulladd f8(m[6],n4,p3[3],c4,n3);
assign m[0]=p0[0];
assign m[1]=s0;
assign m[7]=n4;
endmodule

module lcd1_task4(in_Clk, lcd_rs, lcd_e, data);
input in_Clk;
output reg [7:0] data;
output reg lcd_rs;
output lcd_e;
wire [7:0] command [0:4];
reg [31:0] count=0;
wire out_Clk;
assign command [0] = 8'h38;
assign command [1] = 8'h0C; // keep display on but cursor off
assign command [2] = 8'h06; // increment the cursor
assign command [3] = 8'h01; // clear the display
assign command [4] = 8'h80; // choose the second line
    clk_divider c0 (in_Clk, out_Clk);
assign lcd_e = out_Clk;
always@(posedge lcd_e) begin
    count <= count + 1;
    case(count)
        // Send Initialization Commands (lcd_rs = 0)
        1: begin lcd_rs = 0; data = command[0]; end // 0x38 - Function Set
        2: begin lcd_rs = 0; data = command[1]; end // 0x0C - Display Control
        3: begin lcd_rs = 0; data = command[2]; end // 0x06 - Entry Mode
        4: begin lcd_rs = 0; data = command[3]; end // 0x01 - Clear Display
        5: begin lcd_rs = 0; data = command[4]; end // 0x80 - Set DDRAM Address
        6: begin lcd_rs = 1; data = 8'h31; end // '1'
        7: begin lcd_rs = 1; data = 8'h32; end // '2'
        8: begin lcd_rs = 1; data = 8'h33; end // '3'
        9: begin lcd_rs = 1; data = 8'h34; end // '4'
        10: begin lcd_rs = 1; data = 8'h35; end // '5'
        11: begin lcd_rs = 1; data = 8'h36; end // '6'
        12: begin lcd_rs = 1; data = 8'h37; end // '7'
        13: begin lcd_rs = 1; data = 8'h38; end // '8'
        14: begin lcd_rs = 1; data = 8'h39; end // '9'
        15: begin lcd_rs = 1; data = 8'h41; end // 'A'
        16: begin lcd_rs = 1; data = 8'h42; end // 'B'
        17: begin lcd_rs = 1; data = 8'h43; end // 'C'
        18: begin lcd_rs = 1; data = 8'h44; end // 'D'
        19: begin lcd_rs = 1; data = 8'h45; end // 'E'
        20: begin lcd_rs = 1; data = 8'h46; end // 'F'
        21: begin lcd_rs = 1; data = 8'h47; end // 'G'
        default: begin
            lcd_rs <= 0;
            data <= 8'h02;
        end
    endcase
end
endmodule


module clk_divider (
    input wire clk_in,
    output clk_out
);
    reg clk_out = 0;
    reg [31:0]count = 0;
    always @(posedge clk_in ) begin
        begin
            if (count == 2_499_999) begin
                count = 0;
                clk_out = ~clk_out;
            end else begin
                count = count + 1;
            end
        end
    end
endmodule
