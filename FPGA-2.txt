D flip-flop module:

module dff(
    output reg q,
    output reg qbar,
    input d,
    input rst,
    input clk
);
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            q <= 1'b0;
            qbar <= 1'b1;
        end
        else begin
            q <= d;
            qbar <= ~d;
        end
    end
endmodule

Johnson counter module:

module jsc(input rst, output [3:0]count, input clk);
    wire [3:0]qbar;
    dff f0(count[3], qbar[3], qbar[0], rst, clk);
    dff f1(count[2], qbar[2], count[3], rst, clk);
    dff f2(count[1], qbar[1], count[2], rst, clk);
    dff f3(count[0], qbar[0], count[1], rst, clk);
endmodule

Implementation of Johnson counter on FPGA:

module cuteboi3(Seven_Seg,in_clk,rst);
    input in_clk, rst;
    output [7:0] Seven_Seg;
    wire [3:0] digit;
    wire q0,q1,q2;
    wire q2bar, q1bar, q0bar;
    wire out_clk;
    clk_divider cd0(in_clk, out_clk);
    jsc j(rst, digit[3:0], out_clk);
    decoder dec0( digit[3:0],Seven_Seg [7:0]);
endmodule

module clk_divider(
    input wire clk_in,
    output clk_out
);
    reg clk_out = 0;
    reg [31:0]count = 0;
    always @(posedge clk_in ) begin
        if (count == 24999999) begin
            count = 0;
            clk_out = ~clk_out;
        end else begin
            count = count + 1;
        end
    end
endmodule

Seven segment display decoder module:

module decoder(
    input wire [3:0] in,
    output reg [7:0] seg
);
    always @(*) begin
        case (in)
            4'b0000: seg = 8'b11000000;
            4'b0001: seg = 8'b11111001;
            4'b0010: seg = 8'b10100100;
            4'b0011: seg = 8'b10011000;
            4'b0100: seg = 8'b10011001;
            4'b0101: seg = 8'b10010010;
            4'b0110: seg = 8'b10000010;
            4'b0111: seg = 8'b11111000;
            4'b1000: seg = 8'b10010000;
            4'b1001: seg = 8'b10000010;
            4'b1010: seg = 8'b11000000;
            4'b1011: seg = 8'b11111001;
            4'b1100: seg = 8'b10100100;
            4'b1101: seg = 8'b10011001;
            4'b1110: seg = 8'b10011001;
            4'b1111: seg = 8'b10010010;
            default: seg = 8'b00000000;
        endcase
    end
endmodule
