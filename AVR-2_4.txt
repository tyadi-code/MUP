.org 0x0000
    RJMP INIT
INIT:
    LDI R16, 0xFF
    OUT DDRD, R16
    LDI R16, 0x00
    OUT DDRB, R16
    LDI R16, 0xFF
    OUT PORTB, R16
MAIN_LOOP:
    IN R16, PINB
    COM R16
    MOV R17, R16
    SWAP R17
    ANDI R17, 0x0F
    MOV R18, R16
    ANDI R18, 0x0F
    ADD R17, R18
    OUT PORTD, R17
    RJMP MAIN_LOOP
